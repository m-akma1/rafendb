# FR-08 Scenario 1: Multi-Transaction Crash Recovery (Redo & Undo + Checkpoint)
#
# Kondisi Awal:
#   Tabel `accounts(id INT, owner VARCHAR)` kosong.
#
# Urutan Eksekusi (Interleaved):
#   1. T101 BEGIN
#   2. T101 INSERT (1, "Alice") -> ditulis ke WAL & StorageManager
#   3. T101 COMMIT -> WAL di-flush permanen
#   4. CHECKPOINT -> checkpoint selesai tercatat di WAL & StorageManager di-flush
#   5. T102 BEGIN
#   6. T102 INSERT (2, "Bob") -> ditulis ke WAL & StorageManager
#   7. T102 ABORT (Rollback) -> undo menghapus (2, "Bob") + menulis CLR & ABORT
#   8. T103 BEGIN
#   9. T103 UPDATE (1, "Alice") -> (1, "Alice-V2") -> ditulis ke WAL & COMMIT tanpa flush SM
#  10. T104 BEGIN
#  11. T104 INSERT (3, "Charlie") -> ditulis ke WAL & StorageManager tanpa COMMIT (simulasi crash)
#
# Hasil yang Diharapkan Setelah Restart & recoverFromCrash():
#   - Record (1) == (1, "Alice-V2")  [REDO berhasil]
#   - Record (2) == tidak ada        [ABORT tetap batal]
#   - Record (3) == tidak ada        [UNDO berhasil]
#
# Cara Verifikasi Otomatis:
#   Jalankan RecoveryDriver.runAllTests() -> [TEST] Real Storage Multi-Tx Recovery: PASSED
#   atau scripts\run.cmd driver recovery --run=test-scenarios\recovery\crash-recovery.txt

begin 101
insert 101 accounts 1 Alice
commit 101
checkpoint
begin 102
insert 102 accounts 2 Bob
rollback 102
begin 103
update-wal 103 accounts 1 Alice-V2
commit 103
begin 104
insert 104 accounts 3 Charlie
crash
recover
assert val accounts 1 Alice-V2
assert val accounts 2 null
assert val accounts 3 null
assert exists accounts 1 true
assert exists accounts 2 false
assert exists accounts 3 false
assert status 101 COMMITTED
assert status 102 ABORTED
assert status 103 COMMITTED
assert status 104 ABORTED
